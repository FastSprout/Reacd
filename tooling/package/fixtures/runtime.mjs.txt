import assert from 'node:assert/strict';
import { UnsupportedAdapterComponentError } from '@fastsprout/reacd';
import * as consumer from './dist/consumer.js';

assert.match(consumer.markup, /Packaged UI/);
assert.match(consumer.markup, /Packed action/);
assert.match(consumer.markup, /Packed preference/);
assert.match(consumer.markup, /Packed form/);
assert.match(consumer.markup, /Packed rows/);
assert.match(consumer.markup, /Scoped SDK defaults/);
assert.match(consumer.markup, /Open packed dialog/);
assert.match(consumer.markup, /Packed calendar/);
assert.match(consumer.markup, /Packed date/);
assert.match(consumer.markup, /Packed chart/);
assert.doesNotMatch(consumer.markup, /Closed content/);
assert.match(consumer.markup, /data-adapter="shadcn-ui"/);
assert.match(consumer.explicitMarkup, /Explicit SDK/);
assert.match(consumer.transport, /Packed transport/);
assert.throws(consumer.unsupported, UnsupportedAdapterComponentError);
console.log('Installed declarations, default and explicit SDK SSR passed.');

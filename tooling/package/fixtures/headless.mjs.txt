import assert from 'node:assert/strict';
import { createElement } from 'react';
import { renderToString } from 'react-dom/server';
import { atom } from 'jotai';
import * as engine from '@fastsprout/reacd';
import { mountApplication } from '@fastsprout/reacd/bootstrap';

assert(import.meta.resolve('@fastsprout/reacd-shadcn'));
for (const vendor of ['@fastsprout/reacd-mui', '@mui/material', '@emotion/react']) {
  assert.throws(() => import.meta.resolve(vendor), { code: 'ERR_MODULE_NOT_FOUND' });
}
assert.equal('shadcnAdapter' in engine, false);
assert.equal(typeof window, 'undefined');
assert.equal(typeof mountApplication, 'function');
await assert.rejects(import('@fastsprout/reacd/adapter-runtime/state/store.js'), {
  code: 'ERR_PACKAGE_PATH_NOT_EXPORTED',
});
const html = renderToString(createElement(engine.ReacdProvider, null,
  createElement(engine.Text, null, 'Default SDK SSR')));
assert.match(html, /data-adapter="shadcn-ui"/);
assert.match(html, /Default SDK SSR/);
const value = atom('initial');
const first = engine.createReacdStore();
const second = engine.createReacdStore();
first.set(value, 'isolated');
assert.equal(second.get(value), 'initial');
console.log('Default SDK installs transitively and supports isolated Node SSR.');

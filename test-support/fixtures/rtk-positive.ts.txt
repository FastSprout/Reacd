import {
  asyncThunkCreator,
  buildCreateSlice,
  configureStore,
  createAsyncThunk,
  createSlice,
  TaskAbortError,
} from '@reduxjs/toolkit';
import type { SerializedError } from '@reduxjs/toolkit';

const slice = createSlice({
  name: 'counter',
  initialState: { count: 0 },
  reducers: { increment(state) { state.count += 1; } },
});
export const store = configureStore({ reducer: slice.reducer });
const createTask = createAsyncThunk.withTypes<{
  state: ReturnType<typeof store.getState>;
  dispatch: typeof store.dispatch;
}>();
export const read = createTask('counter/read', async (_: void, api) =>
  api.getState().count,
);
export const aborted: SerializedError = new TaskAbortError(undefined);
const create = buildCreateSlice({ creators: { asyncThunk: asyncThunkCreator } });
export const jobs = create({
  name: 'jobs',
  initialState: 0,
  reducers: creator => ({
    load: creator.asyncThunk.withTypes<{ rejectValue: string }>()(
      async (_: void, api) => api.rejectWithValue('Declined'),
    ),
  }),
});
